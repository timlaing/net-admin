#include <tunables/global>
profile netadmin flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  signal (send, receive) peer=netadmin,
  # S6 initializes its runtime directory and drops privileges during startup.
  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,
  capability kill,
  # /init is a shell script, so it needs read permission as well as execute.
  /init rix,
  /run.sh rix,
  /bin/** rix,
  /usr/bin/** rix,
  /usr/local/bin/** rix,
  /command/** rix,
  /package/** rix,
  /usr/lib/bashio/** rix,
  /usr/local/lib/** mr,
  /etc/** r,
  /etc/cont-init.d/** rix,
  /etc/cont-finish.d/** rix,
  /etc/services.d/** rix,
  /etc/s6-overlay/** rix,
  /run/ rw,
  /run/** rwkix,
  /app/** r,
  /config/** rwk,
  /data/** rwk,
  /tmp/** rw,
  deny /var/run/docker.sock rw,
}
